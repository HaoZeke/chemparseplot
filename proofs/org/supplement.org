#+TITLE: Landfold FES observation identities
#+OPTIONS: toc:nil num:nil

A GP posterior mean is linear in the observations, so the field we
fit is the scientific object. These statements are the Lea-tagged
identities; Lean names match the =label= keys.

#+BEGIN_EXPORT latex
\begin{equation}
  \mathrm{clip}(F,M)=\min(F,M).
  \label{eq:clip}
\end{equation}

\begin{theorem}[Low branch of the value-clip]\label{thm:clip-low}
% lea: formalize label=clip_of_low
% lea: context={Nat.min_eq_left after le_of_lt.}
If $F<M$ then $\mathrm{clip}(F,M)=F$.
\end{theorem}

\begin{theorem}[High branch of the value-clip]\label{thm:clip-high}
% lea: formalize label=clip_of_high
% lea: context={Nat.min_eq_right on InTail.}
If $M\le F$ then $\mathrm{clip}(F,M)=M$.
\end{theorem}

\begin{equation}
  M\le F_1 \;\wedge\; M\le F_2
  \;\implies\;
  \mathrm{clip}(F_1,M)=\mathrm{clip}(F_2,M).
  \label{eq:tail}
\end{equation}

\begin{theorem}[Clip erases tail contrast]\label{thm:tail}
% lea: formalize label=clip_erases_tail_contrast uses={clip_of_high}
% lea: context={Rewrite both sides with clip_of_high.}
If both cells sit in the tail, their clipped values are equal.
\end{theorem}

\begin{theorem}[Clip is not injective on the tail]\label{thm:tail-inj}
% lea: formalize label=clip_not_injective_on_tail uses={clip_erases_tail_contrast}
% lea: context={Witness F1=M and F2=M+1.}
There exist distinct tail ranks that clip identifies.
\end{theorem}

\begin{equation}
  \mathrm{keep}(F,M)\;\iff\; F<M.
  \label{eq:cut}
\end{equation}

\begin{theorem}[Membership cut on the tail]\label{thm:cut}
% lea: formalize label=below_ceiling_of_tail
% lea: context={Nat.not_lt.mpr.}
A tail cell fails the membership cut.
\end{theorem}

\begin{equation}
  \mathrm{lerp}_{n/d}(\mathrm{clip}(F_1,M),\mathrm{clip}(F_2,M))=M
  \quad\text{when both cells are in the tail.}
  \label{eq:lerp-ceil}
\end{equation}

\begin{theorem}[Clipped tail interpolates to the ceiling]\label{thm:lerp-ceil}
% lea: formalize label=clipped_tail_lerp_is_ceiling uses={clip_of_high,lerp_const}
% lea: context={Rewrite clips then lerp_const. Denominator d>0.}
The convex hull of a value-clipped tail is the ceiling.
\end{theorem}

\begin{theorem}[Dropping the ceiling keeps contrast]\label{thm:cut-keep}
% lea: formalize label=drop_ceiling_keeps_contrast uses={below_ceiling_of_tail}
% lea: context={Both cells fail BelowCeiling and remain unequal.}
The membership cut does not identify two distinct tail ranks.
\end{theorem}
#+END_EXPORT

* Finite GP representer

The noise-free posterior mean on $n$ sites is the kernel expansion
$\mu_i=\sum_j K_{ij}\alpha_j$ with $K\alpha=y$. That is the
representation: the field at the observation sites is a section of
the Gram matrix.

#+BEGIN_EXPORT latex
\begin{equation}
  K_{ij}=K_{ji},\qquad K_{ii}\ge 0.
  \label{eq:kernel}
\end{equation}

\begin{theorem}[Kernel symmetry]\label{thm:kernel-sym}
% lea: formalize label=kernel_symmetric
% lea: context={Kernel.sym.}
$K_{ij}=K_{ji}$.
\end{theorem}

\begin{theorem}[Nonnegative diagonal]\label{thm:kernel-diag}
% lea: formalize label=kernel_diag_nonneg
% lea: context={Kernel.diag_nonneg.}
$K_{ii}\ge 0$.
\end{theorem}

\begin{equation}
  \mu_i=\sum_j K_{ij}\alpha_j.
  \label{eq:representer}
\end{equation}

\begin{equation}
  K\alpha=y \;\implies\; \mu_i=y_i.
  \label{eq:interp}
\end{equation}

\begin{theorem}[Representer interpolates the table]\label{thm:interp}
% lea: formalize label=representer_interpolates
% lea: context={The site values of the expansion are the Gram action.}
If $K\alpha=y$ then the representer reproduces every observation.
\end{theorem}

\begin{theorem}[One-site interpolation]\label{thm:interp-one}
% lea: formalize label=representer_interpolates_one uses={representer_interpolates}
% lea: context={finRange 1; $\mu=k\alpha$.}
On one site, $k\alpha=y$ if and only if the representer equals $y$.
\end{theorem}

\begin{theorem}[Representer is linear in the coefficients]\label{thm:linear}
% lea: formalize label=representer_linear
% lea: context={foldl of a pointwise sum; Int.mul_add.}
$\mu[\alpha+\beta]=\mu[\alpha]+\mu[\beta]$.
\end{theorem}

\begin{theorem}[Representer scales]\label{thm:smul}
% lea: formalize label=representer_smul
% lea: context={foldl of a scaled family.}
$\mu[c\alpha]=c\,\mu[\alpha]$.
\end{theorem}

\begin{theorem}[One-site coefficient is unique]\label{thm:unique-one}
% lea: formalize label=representer_unique_one uses={representer_interpolates_one}
% lea: context={$k(\alpha-\beta)=0$ and $k\neq 0$.}
A nonzero one-site kernel has at most one representer coefficient.
\end{theorem}

\begin{equation}
  K\,\mathrm{adj}(K)\,y=(\det K)\,y.
  \label{eq:gram-adj}
\end{equation}

\begin{theorem}[Two-site Cramer reproduction]\label{thm:gram-adj}
% lea: formalize label=gram_adjugate
% lea: context={Expand $K\,\mathrm{adj}(K)$. The identity is $K\,\mathrm{adj}(K)=(\det K)I$.}
The two-site Gram matrix applied to the Cramer adjugate recovers
$(\det K)y$. The noise-free GP mean at the training sites is that
table whenever $\det K\neq 0$.
\end{theorem}

\begin{theorem}[Adjugate from the left]\label{thm:adj-gram}
% lea: formalize label=adjugate_gram uses={gram_adjugate}
% lea: context={Expand $\mathrm{adj}(K)\,K$.}
$\mathrm{adj}(K)\,K\alpha=(\det K)\alpha$.
\end{theorem}

\begin{theorem}[Observation linearity]\label{thm:obs-linear}
% lea: formalize label=obs_linear
% lea: context={adjK2 is linear in y.}
Cramer coefficients of $y+z$ are the sum of the coefficients.
\end{theorem}

\begin{theorem}[Two-site uniqueness]\label{thm:unique-two}
% lea: formalize label=gram_unique uses={adjugate_gram}
% lea: context={$K(\alpha-\beta)=0$, apply adj, $\det\neq 0$.}
A 2-site kernel with nonzero determinant has at most one coefficient pair.
\end{theorem}

\begin{theorem}[Two-site positive determinant]\label{thm:spd-two}
% lea: formalize label=kernel2_det_pos
% lea: context={$b^2 < ac$ rearranges to $\det K>0$.}
If $a>0$, $c>0$ and $b^2<ac$ then $\det K>0$.
\end{theorem}
